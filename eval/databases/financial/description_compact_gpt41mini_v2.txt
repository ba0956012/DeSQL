-- order: bank transfer orders  
-- account: bank accounts  
-- card: credit cards  
-- client: bank clients  
-- disp: account-client relations (dispositions)  
-- district: geographic and socio-economic data  
-- loan: loans linked to accounts  
-- trans: account transactions  

-- disp.type: disposition type ('OWNER' = account owner, 'DISPONENT' = authorized user)  
-- card.type: card class ('junior', 'classic', 'gold')  
-- client.gender: 'F' = female, 'M' = male  
-- account.frequency: fee frequency ('POPLATEK MESICNE' = monthly fee, 'POPLATEK PO OBRATU' = fee per turnover, 'POPLATEK TYDNE' = weekly fee)  
-- loan.status: 'A' = finished OK, 'B' = finished unpaid, 'C' = running OK, 'D' = running in debt  
-- order.k_symbol: payment purpose ('POJISTNE' = insurance, 'SIPO' = household, 'LEASING' = leasing, 'UVER' = loan payment)  
-- trans.type: transaction type ('PRIJEM' = credit, 'VYDAJ' = withdrawal)  
-- trans.operation: transaction mode ('PREVOD NA UCET' = remittance, 'PREVOD Z UCTU' = collection, 'VKLAD' = cash deposit, 'VYBER' = cash withdrawal, 'VYBER KARTOU' = card withdrawal)  
-- trans.k_symbol: transaction purpose ('UROK' = interest, 'UVER' = loan, 'POJISTNE' = insurance, 'SIPO' = household, 'SLUZBY' = services, 'SANKC. UROK' = penalty interest, 'DUCHOD' = pension)  

-- district.a10: % urban inhabitants  
-- district.a11: average salary  
-- district.a12: unemployment rate 1995  
-- district.a13: unemployment rate 1996  
-- district.a14: entrepreneurs per 1000 inhabitants  
-- district.a15: crimes committed 1995  
-- district.a16: crimes committed 1996  

-- order.bank_to and trans.bank: bank codes (e.g., 'YZ', 'ST', 'QR')  
-- order.account_to and trans.account: recipient account number (different from disp.account_id)  

-- disp.type 'OWNER' means client owns the account; 'DISPONENT' means client authorized to operate it  
-- loan.payments = monthly payment amount (unit: USD)  
-- loan.duration = loan length in months  
-- account.date = account creation date  
-- card.issued = card issue date
